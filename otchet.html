Введение

Разработка веб-сайта «Архив показаний счетчиков» представляет собой процесс создания функционального интерфейса для учёта и систематизации показаний приборов учёта коммунальных ресурсов (вода, газ, электричество). В рамках данной лабораторной работы необходимо выполнить вёрстку веб-страниц архива показаний, страницу ввода показаний, страницу авторизации и страницу квитанции.

Основой для работы служит макет, разработанный в Figma. Задача вёрстки — точно воспроизвести визуальное оформление макета, а также обеспечить функциональность всех элементов интерфейса в соответствии с техническим заданием.

Цель данной работы — получить готовый веб-сайт, который соответствует макету дизайна и удовлетворяет всем 30+ функциональным и нефункциональным требованиям. Выполнение этой задачи требует применения современных технологий веб-разработки (HTML5, CSS3, чистый JavaScript) и соблюдения стандартов вёрстки.

Заключение

В ходе выполнения лабораторной работы были успешно реализованы все запланированные веб-страницы сайта «Архив показаний счетчиков» в соответствии с макетом Figma и техническим заданием.

Разработаны и сверстаны:

  • Главная страница (дашборд) с карточками ресурсов, графиком потребления и таблицей последних показаний;
  • Страница архива показаний с фильтрацией, экспортом в Excel/PDF и возможностью печати индивидуальных квитанций;
  • Страница ввода показаний с валидацией и автоматическим расчётом;
  • Страница авторизации;
  • Страницы ошибок 403 и 404;
  • Адаптивные версии для планшетов и мобильных устройств;
  • Версии для печати.

Все функциональные и нефункциональные требования, включая семантическую вёрстку, микроразметку, Open Graph, ЧПУ, настройку robots.txt и sitemap.xml, выполнены в полном объёме.

Данная работа позволила углубить практические навыки в области веб-вёрстки, в частности, в использовании HTML5 для семантического структурирования контента, CSS3 для создания адаптивного и кроссбраузерного дизайна, а также чистого JavaScript для реализации интерактивных элементов без использования фреймворков.

Приложение 2. JavaScript-код

validation.js (валидация и расчёт)

function calculateTotal() {
    const prev = parseFloat(prevReading.value) || 0;
    const curr = parseFloat(currentReading.value) || 0;
    
    if (curr > prev) {
        const consumption = curr - prev;
        consumptionSpan.textContent = consumption.toFixed(3);
        const amount = consumption * currentTariff;
        totalAmountSpan.textContent = amount.toFixed(2);
        currentReading.setCustomValidity('');
        currentError.style.display = 'none';
    } else {
        consumptionSpan.textContent = '0';
        totalAmountSpan.textContent = '0';
        if (currentReading.value) {
            currentReading.setCustomValidity('Показания должны быть больше предыдущих');
            currentError.style.display = 'block';
        }
    }
}

Генерация квитанции для печати

function generateReceiptHTML(rowData) {
    return `<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>Квитанция ${rowData.resource}</title>
    <style>
        body { font-family: 'Courier New', monospace; margin: 20px; }
        table { width: 100%; border-collapse: collapse; }
        th, td { border: 1px solid black; padding: 8px; }
        @media print { .no-print { display: none; } }
    </style>
</head>
<body>
    <div class="receipt">
        <h2>Квитанция об оплате</h2>
        <p>№ Счетчика: ${rowData.meterNumber}</p>
        <table>
            <tr><th>Дата</th><th>Пред.</th><th>Текущ.</th></tr>
            <tr><td>${rowData.date}</td><td>${rowData.prev}</td><td>${rowData.current}</td></tr>
        </table>
        <table>
            <tr><th>Услуга</th><th>Объем</th><th>Тариф</th><th>Сумма</th></tr>
            <tr><td>${rowData.resource}</td><td>${rowData.consumption}</td>
                <td>${rowData.tariff}</td><td>${rowData.amount}</td></tr>
        </table>
        <div class="total">Всего к оплате: ${rowData.amount} руб.</div>
        <div class="no-print">
            <button onclick="window.print()">Печать</button>
        </div>
    </div>
</body>
</html>`;
}

function openReceiptPrint(rowData) {
    const html = generateReceiptHTML(rowData);
    const win = window.open('', '_blank', 'width=800,height=600');
    win.document.write(html);
    win.document.close();
    setTimeout(() => win.print(), 200);
}

charts.js (график на Canvas)

function drawChart(resourceType) {
    const data = chartData[resourceType];
    const values = data.data;
    const maxValue = Math.max(...values) * 1.1;
    
    // Очистка canvas
    chartContext.clearRect(0, 0, width, height);
    
    // Отрисовка сетки
    for (let i = 0; i <= 5; i++) {
        const y = padding.top + chartHeight - (i / 5) * chartHeight;
        chartContext.beginPath();
        chartContext.moveTo(padding.left, y);
        chartContext.lineTo(padding.left + chartWidth, y);
        chartContext.stroke();
    }
    
    // Отрисовка линии
    chartContext.beginPath();
    chartContext.strokeStyle = color;
    values.forEach((value, index) => {
        const x = padding.left + index * (chartWidth / (values.length - 1));
        const y = padding.top + chartHeight - (value / maxValue) * chartHeight;
        if (index === 0) chartContext.moveTo(x, y);
        else chartContext.lineTo(x, y);
    });
    chartContext.stroke();
}
